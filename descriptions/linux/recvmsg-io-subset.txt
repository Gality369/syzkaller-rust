resource fd[4] = -1, 0, 1, 2, 3, 4
resource sock_tx[fd]
resource sock_rx[fd]

const AF_UNIX = 1
const SOCK_DGRAM = 2
const MSG_DONTWAIT = 64
constset zero_4[4] = 0
flagset recv_flags[4] = MSG_DONTWAIT

type iovec_in {
    base ptr[in; array[int8; 4:16]]
    len len[base, intptr]
} [size[16]]

type iovec_out {
    base ptr[out; array[int8; 4:16]]
    len len[base, intptr]
} [size[16]]

type send_msghdr {
    msg_name ptr[in, buffer[16:16], opt]
    msg_namelen len[msg_name, int32]
    msg_iov ptr[in, array[iovec_in; 1:2]]
    msg_iovlen len[msg_iov, intptr]
    msg_control ptr[in, array[int8; 0:16], opt]
    msg_controllen bytesize[msg_control, intptr]
    msg_flags const[zero_4]
} [size[56]]

type recv_msghdr {
    msg_name ptr[out, buffer[16:16], opt]
    msg_namelen len[msg_name, int32]
    msg_iov ptr[in, array[iovec_out; 1:2]]
    msg_iovlen len[msg_iov, intptr]
    msg_control ptr[out, array[int8; 0:16], opt]
    msg_controllen bytesize[msg_control, intptr]
    msg_flags const[zero_4]
} [size[56]]

sock_msg_pair {
    tx sock_tx
    rx sock_rx
} [size[8]]

socketpair(domain const[AF_UNIX], type const[SOCK_DGRAM], proto const[0], fds ptr[out, sock_msg_pair]) int
sendmsg(fd sock_tx, msg ptr[in, send_msghdr], flags const[0]) int
recvmsg(fd sock_rx, msg ptr[inout, recv_msghdr], flags flags[recv_flags]) int
